Pi 是什么？

常说的 Pi 是一个 Coding Agent ，Pi Coding Agent 包含两个主要的部分：

  1. Agent Core，运行 Agent Loop：
    1. 上下文管理：
      1. 系统提示词：
        1. Pi 的内置提示词很少；
        2. 系统提示词更对来源于项目设置。
      2. 项目内容；
      3. 工具列表；
      4. 会话历史；
      5. 用户消息；
      6. skills、extensions 描述。
    2. 调用 LLM；
    3. 处理 Response；
    4. 调用工具：
      1. 文件的读、写、编辑；
      2. 目录检索、内容检索；
      3. 运行 shell 命令。
    5. 将上下文和工具调用结果重新发送给 LLM；
    6. 直到任务完成。
  2. Pi Interactive，在终端处理用户交互和 Agent Core 工作流：
    1. 用户交互，例如 TUI：
      1. Chat Input；
      2. ScrollView；
      3. Stream Output；
      4. Commands；
      5. 会话选择等。
    2. 会话管理，LLM 的调用是无状态的：
      1. 会话历史内容；
      2. 工具调用结果；
      3. 其他和任务执行相关的状态。
    3. 上下文压缩：
      1. contextTokens（已使用） > contextWindow（模型最大支持） - reserveTokens（模型回复、系统开销等保留的；默认 16384）
      2. 上下文到达上限会自动压缩，压缩时保留最近的对话（默认 20000），只压缩更早的部分；
      3. 旧的上下文交给 LLM 总结，保留用户目标、重要决策、修改记录等。
    4. skills、extensions、commands 管理。

/**
 * https://medium.com/@ramunarasinga/undici-setglobaldispatcher-function-in-pi-coding-agent-package-43f538041257
 * 这里使用外部的 undici，而非 Node 内置的 undici：
 * 1. 更多新特性，内置的 fetch 不支持 Socks5 代理、链式代理...
 * 2. 更好的极限性能，可以对 fetch 参数进行修改
 * 3. 手动优化连接池，修改 fetch 连接池策略
 * 4. HTTP 管线化，提升并发数量
 * 5. 和 Node 版本解耦
 */
import * as undici from 'undici';
export function configureHttpDispatcher(
  timeoutMs: number = DEFAULT_HTTP_IDLE_TIMEOUT_MS
): void {
  const normalizedTimeoutMs = parseHttpIdleTimeoutMs(timeoutMs);
  if (normalizedTimeoutMs === undefined) {
    throw new Error(`HTTP 超时配置不合法：${String(timeoutMs)}`);
  }
  /**
   * EventEmitter.prototype.on.call(dispatcher, "error", (_error: unknown): void => {});
   * withUndiciErrorListener 的作用是：防止未被捕获的错误导致进程崩溃
   * 如果一个 EventEmitter 实例触发了 "error" 事件
   * 但该实例上没有挂载任何 "error" 事件监听器
   * 未捕获异常会导致整个进程崩溃退出
   */
  const dispatcher = withUndiciErrorListener(
    /**
     * https://github.com/nodejs/undici/blob/main/docs/docs/api/EnvHttpProxyAgent.md
     * 会读取用户环境的 http_proxy、https_proxy、no_proxy 的代理终端配置
     * 比如，一些代理软件没有走 Tunnel 而代理某个 Port，需要设置这些
     */
    new undici.EnvHttpProxyAgent({
      allowH2: false, // 禁用 HTTP/2
      bodyTimeout: normalizedTimeoutMs, // 多个 bodychunk 的超时时间
      connect: {
        // Node 的默认值是 250ms
        autoSelectFamilyAttemptTimeout: 2_000
      },
      headersTimeout: normalizedTimeoutMs, // 请求的超时时间
      // new undici.Client(...)
      clientFactory: createUndiciClient,
      // new undici.Pool(...)
      factory: createUndiciOriginDispatcher
    })
  );
  /**
   * https://github.com/nodejs/undici#undicisetglobaldispatcherdispatcher
   * setGlobalDispatcher 影响所有使用 undici 的模块，包括 Node 内置的
   * undici 是 fetch 的底层，会影响 fetch 的行为
   * undici 不会影响内置的 http、https 行为
   */
  undici.setGlobalDispatcher(dispatcher);
  /**
   * 确保全局的 fetch 和底层的网络调度器（dispatcher）来自“同一个”undici 代码实现
   * 因为 undici.setGlobalDispatcher 会给所有 fetch 注入 dispatcher
   * dispatcher 不匹配，会导致响应主体解析错误 / 乱码
   * fetch('https://...', { dispatcher: xxx })
   */
  const shouldInstallGlobals =
    installedGlobalFetch === undefined
      ? // 首次执行，判断当前 fetch 是否是 Node.js 默认原生 fetch
        globalThis.fetch === originalGlobalFetch
      : // 非首次执行，判断期间是否被外部改动过
        globalThis.fetch === installedGlobalFetch;
  if (shouldInstallGlobals) {
    /**
     * 安装修改后的 fetch：
     * globalThis.Response = npmUndici.Response;
     * globalThis.Request = npmUndici.Request;
     * globalThis.Headers = npmUndici.Headers;
     * globalThis.fetch = npmUndici.fetch;
     */
    undici.install?.();
    installedGlobalFetch = globalThis.fetch;
  }
}

export async function main(args: string[], options?: MainOptions) {
  // resetTimings 以及 time 主要用来对每个阶段的处理时间做记录
  resetTimings();
  /**
   * packages/coding-agent/src/extensions/llama/index.ts
   * builtInExtensions 是一个名为 llama.cpp 扩展，用来支持本地模型
   * [{ name: "llama.cpp", factory: llamaExtension, hidden: true }];
   * hidden: true 表示他是基础设施的一部分，用户通过界面是查不到他的
   */
  const extensionFactories = [
    ...builtInExtensions,
    ...(options?.extensionFactories ?? [])
  ];
  /**
   * https://pi.dev/docs/latest/settings#telemetry-and-update-checks
   * 在启动时（加载用户设置之前），可以禁用网络活动
   * 包括更新检查、软件包更新检查和安装等
   */
  const offlineMode =
    args.includes('--offline') || isTruthyEnvFlag(process.env.PI_OFFLINE);
  if (offlineMode) {
    process.env.PI_OFFLINE = '1';
    process.env.PI_SKIP_VERSION_CHECK = '1';
  }
  /**
   * runAuthCommand：
   * 1. 判断是否输入了 auth help、--help、-h
   * 2. 使用 parseAuthCommand 判断是否输入了 auth *：
   *    1. 将 auth * 做映射
   *    2. 扫描 flag 并校验
   *    3. 返回封装好的 { kind, args: commandArgs, json, credentials, noRefresh }
   *    4. { kind, args: commandArgs, json, credentials, noRefresh, minExpiryMs }
   *    5. pi auth check --provider openai、pi auth check --model openai/gpt-5.5 --json --credentials 之类的
   * 3. 调用 parseArgs 把命令包装成一个对象 { messages:[], fileArgs:[], unknownFlags: new Map(), diagnostics: [] }
   * 4. 命令不合法 / 执行过程中有报错信息，就 throw；如果命令生效就 return true，流程结束
   */
  if (await runAuthCommand(args)) {
    return;
  }
  // 处理在 Windows 平台下，pi update 可能失败的问题
  if (process.platform === 'win32') {
    cleanupWindowsSelfUpdateQuarantine(getPackageDir());
  }
  const cwd = process.cwd();
  const agentDir = getAgentDir(); // 获取全局配置目录 (例如 ~/.pi/agent/)
  /**
   * 读取全局配置文件并返回 SettingsManager 实例：
   * 1. 读取配置
   * 2. 持久化配置
   * 3. 迁移过时的配置
   * 4. 配置校验
   */
  const bootstrapSettingsManager = SettingsManager.create(cwd, agentDir, {
    projectTrusted: false
  });
  // 如果设置中有配置 http_proxy、https_proxy，更新到环境变量
  applyHttpProxySettings(
    bootstrapSettingsManager.getGlobalSettings().httpProxy
  );
  // 应用配置中的 http_proxy、https_proxy
  configureHttpDispatcher();
  // 处理 Pi 自己的更新以及 pi install / remove / list 等对扩展的操作命令
  if (await handlePackageCommand(args, { extensionFactories })) {
    const exitCode = process.exitCode ?? 0;
    if (
      process.platform === 'win32' &&
      exitCode === 0 &&
      args[0] === 'update'
    ) {
      /**
       * 如果在 fetch 还在 teardown 时调用 process.exit(0)，Node 可能会 assert 崩溃
       * https://github.com/nodejs/node/issues/56645
       */
      return;
    }
    process.exit(exitCode);
    return;
  }
  // 处理 pi config * 相关的命令
  if (await handleConfigCommand(args, { extensionFactories })) {
    return;
  }
  /**
   * 将命令解析为对象
   * {
   *   messages:[],
   *   fileArgs:[],
   *   unknownFlags: new Map(),
   *   diagnostics: []
   * }
   */
  const parsed = parseArgs(args);
  // 解析过程中出现的错误统一抛出
  if (parsed.diagnostics.length > 0) {
    for (const d of parsed.diagnostics) {
      const color = d.type === 'error' ? chalk.red : chalk.yellow;
      console.error(
        color(`${d.type === 'error' ? 'Error' : 'Warning'}: ${d.message}`)
      );
    }
    if (parsed.diagnostics.some(d => d.type === 'error')) {
      process.exit(1);
    }
  }
  time('parseArgs');
  // 打印版本信息
  if (parsed.version) {
    console.log(VERSION);
    process.exit(0);
  }
  // pi --export，将 session 导出
  if (parsed.export) {
    let result: string;
    try {
      const outputPath =
        parsed.messages.length > 0 ? parsed.messages[0] : undefined;
      /**
       * 通过 SessionManager 将 session 导出为 HTML
       * packages/coding-agent/src/core/export-html
       * 例如：https://mariozechner.at/posts/2025-11-30-pi-coding-agent/media/long-session.html
       */
      result = await exportFromFile(parsed.export, outputPath);
    } catch (error: unknown) {
      const message =
        error instanceof Error ? error.message : 'Failed to export session';
      console.error(chalk.red(`Error: ${message}`));
      process.exit(1);
    }
    console.log(`Exported to: ${result}`);
    process.exit(0);
  }
  /**
   * 通过命令 & TTY（表示当前进程是否运行在标准终端环境）推断运行模式：
   * pi --mode rpc  → RPC 模式（用于编辑器插件等外部进程通信）
   * pi --mode json → JSON 输出模式
   * pi -p          → 打印模式
   * pi             → 交互模式
   */
  let appMode = resolveAppMode(
    parsed,
    process.stdin.isTTY,
    process.stdout.isTTY
  );
  const shouldTakeOverStdout =
    appMode !== 'interactive' && !isPlainRuntimeMetadataCommand(parsed);
  /**
   * 交互模式下，使用 TUI 渲染，不用劫持 stdout
   * 裸的 --help 或 --list-models 命令，不用劫持 stdout
   * 为了防止外部组件 / 库将内容 console 到 process.stdout 影响最终输出
   * 将 process.stdout 的实现替换为 process.stderr，实际内容通过 stderr 输出
   */
  if (shouldTakeOverStdout) {
    takeOverStdout();
  }
  if (parsed.mode === 'rpc' && parsed.fileArgs.length > 0) {
    console.error(
      chalk.red('Error: @file arguments are not supported in RPC mode')
    );
    process.exit(1);
  }
  validateForkFlags(parsed); // 校验是否有和 --fork 冲突的 flag
  validateSessionIdFlags(parsed); // 检验是否有和 --session-id 冲突的 flag
  /**
   * 完成一些数据迁移：
   * 1. migrateAuthToAuthJson，处理一些 API Key、API OAuth
   * 2. migrateSessionsFromAgentRoot，处理一些会话记录 .jsonl
   * 3. migrateToolsToBin，将一些二进制文件从 tools/ 迁移到 bin/ 目录
   * 4. migrateKeybindingsConfigFile，迁移一些快捷键绑定 keybindings.json
   */
  const { migratedAuthProviders: migratedProviders, deprecationWarnings } =
    runMigrations(cwd);
  time('runMigrations');
  /**
   * 完成各种迁移之后，重新获取最新的配置，这次没有了 { projectTrusted: false }
   * cwd 是运行目录、agentDir 是系统用户目录下的 .pi/agent
   * SettingsManager.create 会初始化一个 FileSettingsStorage 实例
   * FileSettingsStorage 实例用来控制（加锁）两个目录下的 settings.json：
   * *  写操作：如果文件不存在但要写入，会创建目录后再加锁
   * *  读操作：如果文件存在，先加锁再读
   * SettingsManager.fromStorage 会尝试读取并合并两个目录下的 settings.json
   * 并保存到 SettingsManager.settings 上
   */
  const startupSettingsManager = SettingsManager.create(cwd, agentDir);
  reportDiagnostics(
    collectSettingsDiagnostics(startupSettingsManager, 'startup session lookup')
  );
  // 是不是官方 Pi 在实验模式下的第一次启动
  if (
    appMode === 'interactive' &&
    !parsed.help &&
    parsed.listModels === undefined &&
    shouldRunFirstTimeSetup()
  ) {
    // 可以选主题 & 数据分析偏好
    await showFirstTimeSetup(startupSettingsManager);
    time('firstTimeSetup');
  }
  // 应用主题配置并合并到 SettingsManager.settings 上
  if (appMode === 'interactive' && parsed.useTheme !== undefined) {
    startupSettingsManager.applyOverrides({ theme: parsed.useTheme });
  }
  /**
   * 启动时的 cwd 和 session 对应的目录可能不同：
   * 1. 在创建任何和 cwd 绑定的服务时，确认最终的 runtimeDir
   * 2. --session 和 --resume 可能会在 ./project-a 下选择 project-b 的 sessionId
   * 3. 和项目相关的设置、资源、provider、models 必须在拿到目标 session 的 cwd 之后才能解析
   * 4. startupSettingsManager 只会用来在选择 session 期间进行 sessionDir 查找，不参与最终 runtime 配置
   */
  const envSessionDir = process.env[ENV_SESSION_DIR];
  const sessionDir =
    (parsed.sessionDir ? normalizePath(parsed.sessionDir) : undefined) ??
    (envSessionDir ? expandTildePath(envSessionDir) : undefined) ??
    startupSettingsManager.getSessionDir();
  /**
   * 如果命令中带了 --fork、--session、--resume、--continue
   * 使用 SessionManager 打开对应的 session
   * 简而言之，命令参数比当前目录更重要（觉得前三个连 in-memory 都没内必要）：
   * 1. --no-session        → 创建内存中的临时 session（不保存）
   * 2. --help              → 创建内存中的临时 session（不保存）
   * 3. --list-models       → 创建内存中的临时 session（不保存）
   * 4. --fork <source>     → 从指定 session 派生出新 session
   * 5. --session <path|id> → 按路径或 sessionID 打开指定 session
   * 6. --resume            → 弹出 session 选择器让用户选
   * 7. --continue          → 继续当前项目最近一次的 session
   * 8. --session-id <id>   → 用指定 ID 创建或打开 session
   * 9. 没有任何参数        → 在 `cwd` 下创建全新 session
   */
  let sessionManager = await createSessionManager(
    parsed,
    cwd,
    sessionDir,
    startupSettingsManager
  );
  /**
   * 判断 session 对应的 sessionFile（.jsonl）和工作目录是否还存在
   * 返回 { sessionFile, sessionCwd, fallbackCwd } 或 undefined
   */
  const missingSessionCwdIssue = getMissingSessionCwdIssue(sessionManager, cwd);
  // 如果 session 目录丢失了
  if (missingSessionCwdIssue) {
    if (appMode === 'interactive') {
      /**
       * 交互模式下，让用户选择「继续」还是「取消」
       * 继续，使用当前的 cwd
       * 取消，退出程序
       */
      const selectedCwd = await promptForMissingSessionCwd(
        missingSessionCwdIssue,
        startupSettingsManager
      );
      if (!selectedCwd) {
        process.exit(0);
      }
      sessionManager = SessionManager.open(
        missingSessionCwdIssue.sessionFile!,
        sessionDir,
        selectedCwd
      );
    } else {
      // 非交互模式，不能选择
      console.error(
        chalk.red(new MissingSessionCwdError(missingSessionCwdIssue).message)
      );
      process.exit(1);
    }
  }
  // 可以通过 --name 给会话起一个名字
  if (parsed.name !== undefined) {
    const name = normalizeSessionName(parsed.name);
    if (name === undefined) {
      console.error(chalk.red('Error: --name requires a non-empty value'));
      process.exit(1);
    }
    sessionManager.appendSessionInfo(name);
  }
  time('createSessionManager');
  /**
   * .pi/agent/trust.json 这个全局文件保存着项目信任配置
   * sessionCwd 拿到的是 session 的目录（运行目录 / --session 指定的会话目录）
   * 当通过 --approve / -a 启动 Pi 时，projectTrustOverride 会被设置为 true
   */
  const trustStore = new ProjectTrustStore(agentDir);
  const sessionCwd = sessionManager.getCwd();
  /**
   * https://pi.dev/docs/latest/security#project-trust
   * autoTrustOnReloadCwd 的目的，如果出现了需要信任的资源（.pi 等独属于 Pi 的资源），自动帮他标记为信任，不用再次询问
   * hasTrustRequiringProjectResources：
   *   * 去读 session 下的 .pi 目录中的 skills、extensions、SYSTEM.md...，如果存在，返回 true
   *   * 向上遍历 .agents/skills 目录，但会排出用户的 .agents/skills 目录，找到返回 true
   *   * 其余情况（例如，空的 .pi 目录或者没有 .pi 目录）返回 false
   *   * AGENTS.md、CLAUDE.md 等文件都会加载（除非禁止加载上下文）
   * 用户在一个干净目录（没有任何 `.pi/settings.json`、extensions、skills 等需要信任审查的资源）启动 Pi
   * 此时 `hasTrustRequiringProjectResources(cwd) === false`
   * 用户在会话中途通过 `/config`、安装 extensions、添加 skills 等方式，在当前目录创建了信任敏感资源
   * 在 reload 后，`hasTrustRequiringProjectResources(cwd)` === `true`
   */
  const autoTrustOnReloadCwd =
    parsed.projectTrustOverride === undefined &&
    !hasTrustRequiringProjectResources(sessionCwd)
      ? sessionCwd
      : undefined;
  // 根据命令，确定输出模式
  const trustPromptMode: AppMode =
    parsed.help || parsed.listModels !== undefined ? 'print' : appMode;
  // 记录 cwd 对应的信任结果，因为 createRuntime 可能会被多次执行
  const projectTrustByCwd = new Map<string, boolean>();
  // 获取一些配置路径
  const resolvedExtensionPaths = resolveCliPaths(cwd, parsed.extensions);
  const resolvedSkillPaths = resolveCliPaths(cwd, parsed.skills);
  const resolvedPromptTemplatePaths = resolveCliPaths(
    cwd,
    parsed.promptTemplates
  );
  const resolvedThemePaths = resolveCliPaths(cwd, parsed.themes);
  /**
   * 从头开始构建 session 需要的一切
   * 设置、模型、扩展、skill、theme、工具、认证、项目信任、CLI 参数覆盖等
   * session 和 runtime 的区别：
   * 1. session 是聊天状态和交互逻辑
   * 2. runtime 是支撑 session 运行的完整执行环境
   */
  const createRuntime: CreateAgentSessionRuntimeFactory = async ({
    cwd,
    agentDir,
    sessionManager,
    sessionStartEvent,
    projectTrustContext
  }) => {
    const isInitialRuntime = sessionStartEvent === undefined;
    const projectTrustDiagnostics: AgentSessionRuntimeDiagnostic[] = [];
    const cachedProjectTrust = projectTrustByCwd.get(cwd);
    const hasTrustRequiringResources = hasTrustRequiringProjectResources(cwd);
    /**
     * 如果有需要被信任的资源，但没有明确信任 / 不信任时：
     * * parsed.projectTrustOverride === undefined，用户没有显式传 --approve 或 --no-approve
     * * cachedProjectTrust === undefined，本次 Pi 进程还没对当前 cwd 做过信任决定
     * * hasTrustRequiringResources，当前 cwd 存在需要信任审查的项目资源
     */
    const shouldResolveProjectTrust =
      parsed.projectTrustOverride === undefined &&
      cachedProjectTrust === undefined &&
      hasTrustRequiringResources;
    const projectTrusted = shouldResolveProjectTrust
      ? false // 需要二次确认的情况，先按照不信任执行
      : (cachedProjectTrust ??
        parsed.projectTrustOverride ??
        (!hasTrustRequiringResources || trustStore.get(cwd) === true));
    // 根据项目是否被信任，创建 SettingsManager 实例
    const runtimeSettingsManager = SettingsManager.create(cwd, agentDir, {
      projectTrusted
    });
    /**
     * 创建和当前路径绑定的核心运行时 services：
     * * modelRuntime
     * * settingsManager
     * * resourceLoader（扩展、skill、theme、promptTemplate）
     * * 把扩展注册的 provider 注册到 modelRuntime（modelRuntime.registerProvider）
     * * 刷新本地模型目录（modelRuntime.refresh({ allowNetwork: false })）
     * 创建 AgentSession 之前，需要知道有哪些模型、扩展、工具可用：
     * * 用哪个模型
     * * 启用哪些工具
     * * thinkingLevel 是什么
     * * 有哪些 scopedModels 可以切换
     */
    const services = await createAgentSessionServices({
      cwd,
      agentDir,
      settingsManager: runtimeSettingsManager,
      modelRuntimeSignal: AbortSignal.timeout(15_000),
      extensionFlagValues: parsed.unknownFlags,
      resourceLoaderReloadOptions: shouldResolveProjectTrust
        ? {
            /**
             * 这里表示「需要询问用户是否信任」
             * packages/coding-agent/src/core/resource-loader.ts
             * DefaultResourceLoader.reload 会判断这个方法是否存在
             */
            resolveProjectTrust: async ({ extensionsResult }) => {
              /**
               * 获取项目信任
               * 调用 hasTrustRequiringProjectResources
               * 调用 emitProjectTrustEvent
               */
              const trusted = await resolveProjectTrusted({
                cwd,
                trustStore,
                trustOverride: parsed.projectTrustOverride,
                // 控制默认「信任」行为："ask", "always", "never"
                defaultProjectTrust:
                  startupSettingsManager.getDefaultProjectTrust(),
                /**
                 * extensionsResult 在内部叫做 preTrustExtensions
                 * 用来告诉用户哪些会被信任
                 */
                extensionsResult,
                projectTrustContext:
                  projectTrustContext ??
                  createProjectTrustContext({
                    cwd,
                    mode: isInitialRuntime ? trustPromptMode : appMode,
                    settingsManager: startupSettingsManager,
                    hasUI: isInitialRuntime && trustPromptMode === 'interactive'
                  }),
                onExtensionError: message =>
                  projectTrustDiagnostics.push({ type: 'warning', message })
              });
              projectTrustByCwd.set(cwd, trusted);
              return trusted;
            }
          }
        : undefined, // 这里表示项目已经被信任
      resourceLoaderOptions: {
        additionalExtensionPaths: resolvedExtensionPaths,
        additionalSkillPaths: resolvedSkillPaths,
        additionalPromptTemplatePaths: resolvedPromptTemplatePaths,
        additionalThemePaths: resolvedThemePaths,
        noExtensions: parsed.noExtensions, // --no-extensions
        noSkills: parsed.noSkills, // --no-skills
        noPromptTemplates: parsed.noPromptTemplates, // --no-prompt-templates
        noThemes: parsed.noThemes, // --no-themes
        noContextFiles: parsed.noContextFiles, // --no-context-files
        systemPrompt: parsed.systemPrompt,
        appendSystemPrompt: parsed.appendSystemPrompt,
        extensionFactories
      }
    });
    const { settingsManager, modelRuntime, resourceLoader } = services;
    /**
     * 给用户看的启动报告，包含：提示配置错误、扩展加载失败、settings 解析问题等
     * 并不会发送给 LLM，这些错误会导致程序退出
     */
    const diagnostics: AgentSessionRuntimeDiagnostic[] = [
      ...projectTrustDiagnostics,
      ...services.diagnostics,
      ...collectSettingsDiagnostics(settingsManager, 'runtime creation'),
      ...resourceLoader.getExtensions().errors.map(({ path, error }) => ({
        type: 'error' as const,
        message: `Failed to load extension "${path}": ${error}`
      }))
    ];
    const modelPatterns = parsed.models ?? settingsManager.getEnabledModels();
    // 解析 --provider、--model、--models、默认模型等
    const scopedModels =
      modelPatterns && modelPatterns.length > 0
        ? await resolveModelScope(modelPatterns, modelRuntime, {
            signal: AbortSignal.timeout(15_000)
          })
        : [];
    /**
     * 将用户输入转换转换成 AgentSession 的启动配置：
     * 确定模型、处理 thinking_level、处理 Tools
     */
    const {
      options: sessionOptions,
      cliThinkingFromModel,
      diagnostics: sessionOptionDiagnostics
    } = buildSessionOptions(
      parsed, // CLI 解析结果
      scopedModels, // 解析后的模型列表
      // 是否有已经在运行的模型列表
      sessionManager.buildSessionContext().messages.length > 0,
      modelRuntime, // 模型运行时
      settingsManager
    );
    diagnostics.push(...sessionOptionDiagnostics);
    if (parsed.apiKey) {
      if (!sessionOptions.model) {
        diagnostics.push({
          type: 'error',
          message:
            '--api-key 需要通过 --model, --provider/--model, or --models 指定模型'
        });
      } else {
        /**
         * pi --model openai/gpt-4o --api-key sk-xxx
         * 在不修改配置文件 / 环境变量的情况下，临时指定 API Key
         * 内部会调用 RuntimeCredentials.setRuntimeApiKey
         * RuntimeCredentials 是用来处理运行时（非持久化）凭证的
         * 内部同时也会调用 ModelRuntime.synchronizeCredentialState
         * 重新组装 Provider，调用 Model.refresh 获取可用的模型
         * 内部的这些操作都被放在一个队列中，保证状态的一致性
         */
        await modelRuntime.setRuntimeApiKey(
          sessionOptions.model.provider,
          parsed.apiKey
        );
      }
    }
    /**
     * 通过 service 创建 AgentSession（调用 createAgentSession）
     * session 的创建与 service 的创建分离：
     * * service：cwd 绑定的基础设施（模型、设置、资源）
     * * session：对话状态和交互逻辑
     */
    const created = await createAgentSessionFromServices({
      services,
      sessionManager,
      sessionStartEvent,
      model: sessionOptions.model,
      thinkingLevel: sessionOptions.thinkingLevel,
      scopedModels: sessionOptions.scopedModels,
      tools: sessionOptions.tools,
      excludeTools: sessionOptions.excludeTools,
      noTools: sessionOptions.noTools,
      customTools: sessionOptions.customTools
    });
    const cliThinkingOverride =
      parsed.thinking !== undefined || cliThinkingFromModel;
    if (created.session.model && cliThinkingOverride) {
      created.session.setThinkingLevel(created.session.thinkingLevel);
    }
    return {
      ...created,
      services,
      diagnostics
    };
  };
  time('createRuntime');
  /**
   * 调用 createRuntime 并包装成 AgentSessionRuntime 实例
   * AgentSessionRuntime 实例用来管理 session 的生命周期，包含了：
   * * _session、_services、_diagnostics、_modelFallbackMessage
   * * switchSession、newSession、fork、importFromJsonl、dispose
   * * 保存 createRuntime 本身，因为一些方法涉及到了 runtime 重建
   */
  const runtime = await createAgentSessionRuntime(createRuntime, {
    cwd: sessionManager.getCwd(),
    agentDir,
    sessionManager
  });
  time('createAgentSessionRuntime');
  const { services, session, modelFallbackMessage } = runtime;
  const { settingsManager, modelRuntime, resourceLoader } = services;
  /**
   * 重新设置 http_proxy、https_proxy
   * 相比于第一次，这里已经拿到了全部的用户配置
   * 如果一开始没有配置 proxy，项目本地配置了 proxy
   * http_proxy 实际优先级是：
   * 1. shell 环境变量，永远最高，Pi 不覆盖
   * 2. globalSettings 里的 http_proxy → 第一次调用就写进 env
   * 3. projectSettings 里的 http_proxy → 基本被 global 挡住（??=）
   */
  applyHttpProxySettings(settingsManager.getGlobalSettings().httpProxy);
  // 同理，根据配置重新调整 HTTP 底层配置
  configureHttpDispatcher(settingsManager.getHttpIdleTimeoutMs());
  /**
   * 当命令带有 --help，需要将生效的 extensions 的 flags 一并打印出来
   * 而非仅仅打印内部的命令 flag
   */
  if (parsed.help) {
    const extensionFlags = resourceLoader
      .getExtensions()
      .extensions.flatMap(extension => Array.from(extension.flags.values()));
    printHelp(extensionFlags);
    process.exit(0);
  }
  if (parsed.listModels !== undefined) {
    const searchPattern =
      typeof parsed.listModels === 'string' ? parsed.listModels : undefined;
    await listModels(modelRuntime, searchPattern, AbortSignal.timeout(15_000));
    process.exit(0);
  }
  // 对运行模式再次判断，不在终端就切换到 print 输出
  let stdinContent: string | undefined;
  if (appMode !== 'rpc') {
    stdinContent = await readPipedStdin();
    /**
     * 例如 echo "hello" | pi -p @prompt.md、cat prompt.md | pi -p
     * 此时，stdinContent 存在值，进入 print 模式
     */
    if (stdinContent !== undefined && appMode === 'interactive') {
      appMode = 'print';
    }
  }
  time('readPipedStdin');
  /**
   * 例如 pi @prompt.md @screenshot.png "What color is the sky?"
   * 内部会通过 processFileArguments 进行文件的处理：
   * * 图片被转换为 Uint8Array
   * * 其他文件当作文本来处理
   * prepareInitialMessage 会构造第一条给 LLM 的 message
   * 如果单纯输入 pi，initialMessage 就为空
   */
  const { initialMessage, initialImages } = await prepareInitialMessage(
    parsed,
    settingsManager.getImageAutoResize(),
    stdinContent
  );
  time('prepareInitialMessage');
  initTheme(settingsManager.getTheme(), appMode === 'interactive');
  time('initTheme');
  // 比如，缺少依赖、缺少模型、弃用的 API 等
  if (appMode === 'interactive' && deprecationWarnings.length > 0) {
    await showDeprecationWarnings(deprecationWarnings);
  }
  time('resolveModelScope');
  reportDiagnostics(runtime.diagnostics);
  if (runtime.diagnostics.some(diagnostic => diagnostic.type === 'error')) {
    if (
      runtime.diagnostics.some(diagnostic =>
        diagnostic.message.includes('Failed to load extension')
      )
    ) {
      console.error(chalk.yellow(EXTENSION_LOAD_FAILURE_HINT));
    }
    process.exit(1);
  }
  time('createAgentSession');
  if (appMode !== 'interactive' && !session.model) {
    console.error(chalk.red(formatNoModelsAvailableMessage()));
    process.exit(1);
  }
  // 例如，执行 PI_STARTUP_BENCHMARK=1 pi
  const startupBenchmark = isTruthyEnvFlag(process.env.PI_STARTUP_BENCHMARK);
  if (startupBenchmark && appMode !== 'interactive') {
    console.error(
      chalk.red('Error: PI_STARTUP_BENCHMARK only supports interactive mode')
    );
    process.exit(1);
  }
  // RPC 模式下，刷新模型列表
  if (!offlineMode && appMode === 'rpc') {
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 15_000);
    void modelRuntime
      .refresh({ signal: controller.signal })
      .catch(() => {})
      .finally(() => clearTimeout(timeout));
  }
  if (appMode === 'rpc') {
    printTimings();
    await runRpcMode(runtime);
  } else if (appMode === 'interactive') {
    const interactiveMode = new InteractiveMode(runtime, {
      // 迁移到 auth.json 的 Providers
      migratedProviders,
      // packages/coding-agent/src/core/sdk.ts:198-206：
      // session 之前使用的模型用不了的警告信息
      modelFallbackMessage,
      autoTrustOnReloadCwd,
      initialMessage,
      initialImages,
      initialMessages: parsed.messages,
      verbose: parsed.verbose,
      tuiMode: parsed.tuiMode, // "regular" | "fullscreen"
      initialThemeSetting: parsed.useTheme
    });
    if (startupBenchmark) {
      /**
       * 创建界面组件、设置终端、加载主题、初始化 keybinding、设置 stdin handler 等
       * 做完这些，TUI 已经“准备就绪”，但不进入主循环
       */
      await interactiveMode.init();
      time('interactiveMode.init');
      /**
       * 等 150ms 是为了让终端有时间回复，这样 benchmark 更接近真实启动状态
       * TUI 初始化时会向终端发送一些查询：
       * * Kitty Keyboard Protocol 能力查询
       * * 单元格大小查询
       * * 设备属性查询
       */
      await new Promise(resolve => setTimeout(resolve, 150));
      interactiveMode.stop();
      stopThemeWatcher();
      printTimings();
      if (process.stdout.writableLength > 0) {
        await new Promise<void>(resolve =>
          process.stdout.once('drain', resolve)
        );
      }
      if (process.stderr.writableLength > 0) {
        await new Promise<void>(resolve =>
          process.stderr.once('drain', resolve)
        );
      }
      return;
    }
    printTimings();
    // 进入事件循环：等待用户输入、处理消息、渲染更新
    await interactiveMode.run();
  } else {
    printTimings();
    const exitCode = await runPrintMode(runtime, {
      mode: toPrintOutputMode(appMode),
      messages: parsed.messages,
      initialMessage,
      initialImages
    });
    stopThemeWatcher();
    restoreStdout();
    if (exitCode !== 0) {
      process.exitCode = exitCode;
    }
    return;
  }
}

main 函数做了这几件事情：

  • 合并内置扩展（builtInExtensions，目前只有 llama.cpp）与用户通过 pi --extension ./my-extension.ts 传入的扩展工厂；
  • 解析并执行 auth check、auth print-api-key、auth print-bearer-token；
  • 读取全局配置，生成 bootstrapSettingsManager；
  • 读取并应用 http_proxy、https_proxy 配置；
  • 处理 Pi 自身更新及包管理命令以及 pi update、pi install、pi remove、pi list 等；
  • 处理 pi config *；
  • 将命令行参数解析为 Args 对象；
  • 处理 --version、--export、--help…；
  • 检查 --fork / --session-id 与其他 session 相关 flag 的互斥性；
  • 迁移旧数据 / 旧规范的目录结构以及配置；
  • 读取 settings.json 和迁移后的数据，生成 startupSettingsManager；
  • 处理 --session-dir、process.env.PI_SESSION_DIR、settings.json 的关系，确定 sessionDir；
  • 处理 --session、--resume、--continue、--fork、--session-id、--no-session 等关系，创建 sessionManager；
  • 处理 session 的 cwd 丢失的问题（session 还在，项目没了）；
  • 处理项目信任配置：
    • 读写 ~/.pi/agent/trust.json；
    • 根据 --approve / --no-approve、缓存、trust.json、是否存在需审查资源决定 projectTrusted。
  • 读取项目目录内的 extensions、skills、promptTemplate；
  • 创建 AgentSessionRuntime：AgentSession + cwd-bound services；
  • 应用项目中的 http_proxy、https_proxy 配置；
  • 合并 stdinContent、@file 内容，生成 initialMessage；

  constructor(
    runtimeHost: AgentSessionRuntime,

     * 在旧 session 真正销毁前，同步地拆掉扩展相关的 UI 组件，避免 stale extension context 被继续使用
     * session_shutdown 事件处理完后，但当前 session 被 dispose() 之前，同步执行一段清理逻辑
     * AgentSessionRuntime.dispose 方法会：
     * * 向 extensions 提交 session_shutdown 事件
     * * 调用 beforeSessionInvalidate，用于清理 UI
     * * 调用 session.dispose，移除所有事件监听，session 完全结束
     */
    this.runtimeHost.setBeforeSessionInvalidate(() => {
      /**
       * 把 extension 提供的 component 清除掉（可以让 Pi 自己写一个 component）：
       * * 扩展上下文被 dispose 后失效
       * * 但 UI 上还残留着指向旧扩展组件的引用
       * * 用户可能看到空白 / 错误的组件，或触发已释放对象的错误
       */
      this.resetExtensionUI();
    });
    this.runtimeHost.setRebindSession(async () => {
      /**
       * 用户可以在不退出 Pi 的情况下，切换 session：
       * * 把 TUI 的当前 session 引用换成新的
       * * 重新渲染 / 订阅新 session 的事件
       * * 重新应用主题
       * 这里仅仅注册了一个切换 session 的工具函数，会在以下场景使用：
       * * /fork   → fork
       * * /import → importFromJsonl
       * * /new    → newSession
       * * /resume → switchSession
       * 初始化的 session 是在 init / run 方法里绑定的
       */
      await this.rebindCurrentSession({ renderBeforeBind: true });
      await this.themeController.applyFromSettings();
    });

     *         ↑↓ 系统 TTY / PTY
     * process.stdin / process.stdout
     *         ↑↓

     * * *     Component → UI 组件的最小接口：会渲染、能处理输入、能失效重绘

  // ...
  private registerSignalHandlers(): void {
    this.unregisterSignalHandlers();
    /**
     * SIGTERM 和 SIGHUP 是 Unix / Linux 操作系统中用于进程间通信（IPC）的系统信号（POSIX Signals）
     * 当操作系统或用户需要干预、终止或通知某个正在运行的进程时，就会向该进程发送这些信号
     */
    const signals: NodeJS.Signals[] = ['SIGTERM'];
    if (process.platform !== 'win32') {
      signals.push('SIGHUP');
    }
    for (const signal of signals) {
      const handler = () => {
        /**
         * 内部对已经分离的子进程进行了保存，trackedDetachedChildPids
         * * spawn 调用 taskkill 对 PID 清理
         * 调用 shutdown：
         * * 通知 extension
         * * 保存 session
         */
        killTrackedDetachedChildren();
        void this.shutdown({ fromSignal: true });
      };
      process.prependListener(signal, handler);
      this.signalCleanupHandlers.push(() => process.off(signal, handler));
    }
    // 错误处理
    const terminalErrorHandler = (error: Error) => {
      if (isDeadTerminalError(error)) {
        this.emergencyTerminalExit();
      }
      throw error;
    };
    process.stdout.on('error', terminalErrorHandler);
    process.stderr.on('error', terminalErrorHandler);
    this.signalCleanupHandlers.push(() =>
      process.stdout.off('error', terminalErrorHandler)
    );
    this.signalCleanupHandlers.push(() =>
      process.stderr.off('error', terminalErrorHandler)
    );
    // 重置 Terminal、防止停留在 raw mode
    const uncaughtExceptionHandler = (error: Error) =>
      this.uncaughtCrash(error);
    process.prependListener('uncaughtException', uncaughtExceptionHandler);
    this.signalCleanupHandlers.push(() =>
      process.off('uncaughtException', uncaughtExceptionHandler)
    );
  }

    if (this.isInitialized) return;
    this.registerSignalHandlers();
    this.changelogMarkdown = this.getChangelogForDisplay();
    if (
      this.session.scopedModels.length > 0 &&
      (this.options.verbose || !this.settingsManager.getQuietStartup())
    ) {
      // 通过 --verbose 或 quietStartup: false（默认）启动...
    }
    this.renderWidgets(); // 渲染 extension widgets
    this.transcriptScrollView = new TuiLayouts.ScrollView(
      this.documentContainer,
      {
        follow: 'end',
        primary: true,
        overscroll: 'chain',
        scrollbar: this.settingsManager.getFullscreenScrollbar(),
        scrollbarStyle: text => theme.bg('scrollbarThumb', text)
      }
    );

     * pi-tui 提供的一个垂直堆叠布局组件，相当于终端里的 flexbox 纵向布局
     * * 过滤不可见子组件
     * * 让每个子组件按可用宽度渲染

     * * 垂直拼接所有子组件的输出，中间加 gap 空行

     * │   (basis: 0, grow: 1)       │
     * │                             │
     * ├─────────────────────────────┤
     * │  pending messages           │  ← 等待发送的消息

    await this.updateAvailableProviderCount();

      initialMessages
    } = this.options;
    if (migratedProviders && migratedProviders.length > 0) {
      this.showWarning(
        `Migrated credentials to auth.json: ${migratedProviders.join(', ')}`
      );
    }
    const modelsJsonError = this.session.modelRuntime.getError();
    if (modelsJsonError) {
      this.showError(`models.json error: ${modelsJsonError}`);
    }
    if (modelFallbackMessage) {
      this.showWarning(modelFallbackMessage);
    }
    void this.maybeWarnAboutAnthropicSubscriptionAuth();
    /**
     * 给 Agent 发提示词：
     * * extension commands
     * * file-based prompt
     * 在主循环开始之前，就将 initialMessage / initialMessages 发出去了
     * 因为 initialMessage / initialMessages 是用户在启动 pi 时
     * 就在命令上带上去的：pi "What color is the sky?"
     */
    if (initialMessage) {
      try {
        await this.session.prompt(initialMessage, { images: initialImages });
      } catch (error: unknown) {
        const errorMessage =
          error instanceof Error ? error.message : 'Unknown error occurred';
        this.showError(errorMessage);
      }
    }
    if (initialMessages) {
      for (const message of initialMessages) {
        try {
          await this.session.prompt(message);
        } catch (error: unknown) {
          const errorMessage =
            error instanceof Error ? error.message : 'Unknown error occurred';
          this.showError(errorMessage);
        }
      }
    }
    /**
     * 开启主交互循环，拿到用户消息，发给 Agent
     * * 等待用户输入 → 拿到输入 → 调用 session.prompt → 等 AI 回复 → 再次等待用户输入 → ...
     * 为什么要使用 while？this.defaultEditor.onSubmit 不挺好用吗？
     * 1. while + await 将程序包装成一个单线程阻塞式的 Promise，主循环就变成了极度清晰的线性单任务管道
     * 2. 用户快速按多次 Enter，多个 `session.prompt()` 会并发执行，而不是排队顺序执行
     * 3. 很难处理初始消息、流式输出中的 steer、followUp 等复杂流程
     * 4. 没有清晰的地方统一做错误处理
     */
    while (true) {
      /**
       * this.defaultEditor.onSubmit 绑定回调函数中：
       * 1. 会处理 /settings、/model、/export、/import 等内置指令
       * 2. 会处理以 ! 开头的原生 shell 命令
       * 3. 如果上下文正在压缩，压缩完再发送用户内容
       * -------------------------------------------
       * 4. 如果 AI 正在输出内容，就：
       * * * 正常流程：
       * * * * 用户：问题 A
       * * * *   AI：回答 A
       * * * * 用户：问题 B
       * * * *   AI：回答 B
       * * * --------------
       * * * steer 流程：
       * * * * 用户：写个快排
       * * * *   AI：好的，快速排序是一种...（还在输出）
       * * * * 用户：用 Python
       * * * *   AI：...用 Python 实现的算法如下...
       * * * 用户内容被当作 steeringMessage 引导当前内容输出
       * * * steer 就是把用户新输入当作对当前生成的实时修正 / 补充
       * ---------------------------------------------------------
       * 5. 会修改 pendingUserInputs 的值
       * 6. 调用 onInputCallback 传入内容
       */
      const userInput = await this.getUserInput();
      try {
        /**
         * 将用户输入发动给 Agent：
         * - 处理 extension commands (通过 pi.registerCommand 注册的)，在流式输出期间，也会立即处理，不走 LLM
         * - 触发扩展 input 事件，让扩展有机会拦截或转换用户输入
         * - 如果是通过 /skill:name xxx 发送的 prompt，消息会被追加在 <skill name="${skill.name}" location="${skill.filePath}">...</skill>`后边
         * - 如果 promptTemplate 存在，消息会被 promptTemplate 处理
         * - 在流式输出期间，根据 streamingBehavior，处理 steering 和 followUp 消息（判断放在哪个队列里）
         * - 在发送给 LLM 之前验证模型配置
         * - 判断是否需要进行上下文压缩
         * - 构建用户消息对象 [{ type: "text", text: expandedText }]
         * - 触发扩展 before_agent_start 事件
         * - 调用 _runAgentPrompt 将处理过的消息发出去
         */
        await this.session.prompt(userInput);
      } catch (error: unknown) {
        const errorMessage =
          error instanceof Error ? error.message : 'Unknown error occurred';
        this.showError(errorMessage);
      }
    }
  }
  // ...
}

export class AgentSession {
  // ...
  /**
   * 处理通过 pi.registerCommand 注册的 extension 指令
   * 基于 promptTemplates 构建 prompt
   * 如果正在流式输出，将消息加入 steering / followUp 消息队列
   * 在正式发送前，验证 Model 配置
   */
  async prompt(text: string, options?: PromptOptions): Promise<void> {
    const expandPromptTemplates = options?.expandPromptTemplates ?? true;
    const preflightResult = options?.preflightResult;
    let messages: AgentMessage[] | undefined;
    try {
      // 使用指令来处理消息
      if (expandPromptTemplates && text.startsWith('/')) {
        const handled = await this._tryExecuteExtensionCommand(text);
        if (handled) {
          // 表示消息已经被预处理过
          preflightResult?.(true);
          return;
        }
      }
      if (this._compactionAbortController !== undefined) {
        throw new Error('请等待上下文压缩完毕之后再试');
      }
      // 触发 extension 的 input 钩子
      let currentText = text;
      let currentImages = options?.images;
      if (this._extensionRunner.hasHandlers('input')) {
        const inputResult = await this._extensionRunner.emitInput(
          currentText,
          currentImages,
          options?.source ?? 'interactive',
          this.isStreaming ? options?.streamingBehavior : undefined
        );
        if (inputResult.action === 'handled') {
          preflightResult?.(true);
          return;
        }
        if (inputResult.action === 'transform') {
          // 消息被 extension 重写 / 优化过
          currentText = inputResult.text;
          currentImages = inputResult.images ?? currentImages;
        }
      }
      // 处理 skills 和 promptTemplates
      let expandedText = currentText;
      if (expandPromptTemplates) {
        expandedText = this._expandSkillCommand(expandedText);
        expandedText = expandPromptTemplate(expandedText, [
          ...this.promptTemplates
        ]);
      }
      if (this.isStreaming) {
        if (!options?.streamingBehavior) {
          throw new Error('正在流式输出，需要明确放入哪个队列');
        }
        if (options.streamingBehavior === 'followUp') {
          await this._queueFollowUp(expandedText, currentImages);
        } else {
          await this._queueSteer(expandedText, currentImages);
        }
        preflightResult?.(true);
        return;
      }
      /**
       * 上一轮 Agent 执行 bash 的结果可能还暂存在队列里
       * 这里，确保上一轮 Agent 产生的所有 bash 结果，在下一轮开始之前已经进入上下文
       * ---------------------------------------------------------------------------
       *    用户：列出当前目录
       *           ↓
       *    session.prompt("列出当前目录")
       *           ↓
       *    Agent 开始推理 → 决定调用 bash("ls")
       *           ↓
       *    bash 执行完成，但 Agent 还在流式输出
       *           ↓
       *    bash 结果暂存到 _pendingBashMessages（不立即写入上下文）
       *           ↓
       *    Agent 这一轮结束
       *           ↓
       *    用户：查看 README
       *           ↓
       *    session.prompt("查看 README")
       *           ↓
       *    _flushPendingBashMessages() ← 先把上一轮暂存的 ls 结果写进去
       *           ↓
       *    发送新 prompt
       * ----------------------------
       * 流式输出期间，消息顺序很敏感
       * tool_use 和 tool_result 必须成对出现
       * （每个 toolCall 必须有对应的 toolResult，
       * 不强调 toolCall 必须紧接着自己的 toolResult）
       * 如果 bash 在 Agent 还在输出时完成了，立刻把 tool_result 写入 messages 会破坏这个顺序
       * 所以，先放进 _pendingBashMessages，等安全时机再写
       * -------------------------------------------------
       *    用户：ls
       *        ↓
       *    Agent: 执行 bash 命令 ls
       *        ↓
       *    bash 结果：[暂存]
       *        ↓
       *    用户：cat README
       *        ↓
       *    session.prompt("cat README") ← 如果这里不 flush，ls 的结果还没进上下文
       *        ↓
       *    Agent 看到的上下文就没有 ls 的输出，回复会出错
       */
      this._flushPendingBashMessages();
      if (!this.model) {
        throw new Error(formatNoModelSelectedMessage());
      }
      const hasConfiguredAuth =
        this._modelRuntime.hasConfiguredAuth(this.model.provider) ||
        (await this._modelRuntime.checkAuth(this.model.provider)) !== undefined;
      if (!hasConfiguredAuth) {
        const isOAuth = this._modelRuntime.isUsingOAuth(this.model.provider);
        if (isOAuth) {
          throw new Error(
            `"${this.model.provider}" 认证失败` +
              `凭证过期了或者网络出问题了` +
              `运行 '/login ${this.model.provider}' 指令来重新认证`
          );
        }
        throw new Error(formatNoApiKeyFoundMessage(this.model.provider));
      }
      /**
       * 判断需不需要压缩上下文，需要 AssistantMessage 做为依据
       * 当然，不同的 Provider 的返回时机不同、字段也不同，pi-ai 做了处理
       * usage: {
       *   input: number;
       *   output: number;
       *   totalTokens: number;
       *   cacheRead: number;
       *   cacheWrite: number;
       * }
       */
      const lastAssistant = this._findLastAssistantMessage();
      if (lastAssistant) {
        /**
         * 因为 AssistantMessage 通常包含详细的 token 用量信息
         * AssistantMessage 也会包含 stopReason：
         * * 如果 stopReason 是「用户中断」，跳过
         * * 如果 stopReason 是「上下文溢出」，压缩
         * 同时，还会判断 AssistantMessage 的时间和上次压缩的时间
         * * 如果比 compactionEntry.timestamp 还早，说明 AssistantMessage 压缩过了
         * 总之，因为 AssistantMessage 里有许多信息可以作为判断「是否应该压缩」的依据：
         * * 上下文用量 / 大小
         * * 模型配置
         * * 是否压缩过了
         * * 上下文有没有溢出
         */
        await this._checkCompaction(lastAssistant, false);
      }
      messages = [];
      const userContent: (TextContent | ImageContent)[] = [
        { type: 'text', text: expandedText }
      ];
      if (currentImages) {
        userContent.push(...currentImages);
      }
      messages.push({
        role: 'user',
        content: userContent,
        timestamp: Date.now()
      });
      for (const msg of this._pendingNextTurnMessages) {
        /**
         * customMessage 是区别于 steerMessage / followUpMessage：
         * * 用来表示：一个内部事件、一个状态标记、自定义渲染的提示信息
         * * 不属于 (user|assistant|tool)Message 范畴
         * * 前者是 extensions 注入的
         * * 后两者来源于用户输入
         */
        messages.push(msg);
      }
      this._pendingNextTurnMessages = [];
      // 触发 before_agent_start 钩子
      const result = await this._extensionRunner.emitBeforeAgentStart(
        expandedText,
        currentImages,
        this._baseSystemPrompt,
        this._baseSystemPromptOptions
      );
      if (result?.messages) {
        for (const msg of result.messages) {
          // 证明 customMessage 是来自 extensions 的
          messages.push({
            role: 'custom',
            customType: msg.customType,
            content: msg.content ?? [],
            display: msg.display,
            details: msg.details,
            timestamp: Date.now()
          });
        }
      }
      // extensions 可能修改 systemPrompt
      if (result?.systemPrompt !== undefined) {
        this._systemPromptOverride = result.systemPrompt;
        this.agent.state.systemPrompt = result.systemPrompt;
      } else {
        this._systemPromptOverride = undefined;
        this.agent.state.systemPrompt = this._baseSystemPrompt;
      }
    } catch (error) {
      preflightResult?.(false);
      throw error;
    }
    if (!messages) {
      return;
    }
    preflightResult?.(true);
    // userInput 最终被转换为了 messages 被发出去
    await this._runAgentPrompt(messages);
  }
  private async _runAgentPrompt(
    messages: AgentMessage | AgentMessage[]
  ): Promise<void> {
    this._isAgentRunActive = true;
    try {
      // 调用 Agent.runPromptMessages(messages)
      await this.agent.prompt(messages);
      while (await this._handlePostAgentRun()) {
        /**
         * 如果是 assistantMessage：
         * * 检查 steeringQueue 和 followUpQueue 是否为空
         * * 调用 Agent.runPromptMessages(messages)
         * 如果不是 assistantMessage：
         * * 调用 runAgentLoopContinue
         * * runAgentLoopContinue 相比于 runAgentLoop：
         * * * 会直接复用 currentContext
         * * * 不会重新构建新的 Context
         */
        await this.agent.continue();
      }
    } finally {
      this._systemPromptOverride = undefined;
      this._flushPendingBashMessages();
      await this._emitAgentSettled();
    }
  }
  private async _handlePostAgentRun(): Promise<boolean> {
    const msg = this._lastAssistantMessage;
    this._lastAssistantMessage = undefined;
    if (!msg) {
      return false;
    }
    /**
     * 判断最后一条消息的错误原因：
     * * 上下文溢出不算，交给上下文压缩
     * * 服务过期了、欠费了、超额了不算
     * * 服务端错误、500，可以重试
     * * 频率太频繁、429，可以重试
     */
    if (this._isRetryableError(msg) && (await this._prepareRetry(msg))) {
      return true;
    }
    if (
      msg.stopReason === 'error' && // 生成过程中出错
      this._retryAttempt > 0
    ) {
      this._emit({
        type: 'auto_retry_end',
        success: false,
        attempt: this._retryAttempt,
        finalError: msg.errorMessage
      });
      this._retryAttempt = 0;
    }
    // 进行上下文压缩，再试
    if (await this._checkCompaction(msg)) {
      return true;
    }
    /**
     * 虽然 await this.agent.prompt(messages); 已经耗空了 steering/followUpQueue
     * 但 agent_end 会触发一些 handler 的执行，可能会产生一些新的待处理的消息
     * hasQueuedMessages 会重新检查一下 steeringQueue 和 followUpQueue
     */
    return this.agent.hasQueuedMessages();
  }
  // ...
}

export async function runAgentLoop(
  prompts: AgentMessage[],
  context: AgentContext,
  config: AgentLoopConfig,
  emit: AgentEventSink,
  signal: AbortSignal | undefined,
  streamFn: StreamFn
): Promise<AgentMessage[]> {
  const newMessages: AgentMessage[] = [...prompts];
  /**
   * 因为 LLM 调用是无状态的
   * 所以需要和之前的上下文进行合并
   */
  const currentContext: AgentContext = {
    ...context,
    messages: [...context.messages, ...prompts]
  };
  // 触发生命周期钩子函数
  await emit({ type: 'agent_start' }); // Agent 开始处理
  await emit({ type: 'turn_start' }); // 当前这一轮「LLM + 工具调用」开始
  for (const prompt of prompts) {
    // 每条消息都要触发钩子函数
    await emit({ type: 'message_start', message: prompt });
    await emit({ type: 'message_end', message: prompt });
  }
  await runLoop(
    currentContext,
    newMessages,
    config,
    signal,
    emit,
    streamFn ?? getDefaultStreamFn()
  );
  return newMessages;
}
async function runLoop(
  initialContext: AgentContext,
  newMessages: AgentMessage[],
  initialConfig: AgentLoopConfig,
  signal: AbortSignal | undefined,
  emit: AgentEventSink,
  streamFunction: StreamFn
): Promise<void> {
  let currentContext = initialContext;
  let config = initialConfig;
  let firstTurn = true;
  // 检查 steeringQueue
  let pendingMessages: AgentMessage[] =
    (await config.getSteeringMessages?.()) || [];
  // 当 Agent 调用完成，继续消耗 followUpQueue
  while (true) {
    let hasMoreToolCalls = true;
    /**
     * 处理工具调用和 steerMessages：
     * * 当没有更多工具调用并且 steerMessages 被耗尽时，本轮任务的 LLM Loop 才会退出
     * * stopReason === 'error' || stopReason === 'aborted' 算是异常 / 主动退出
     * * 虽然 shouldStopAfterTurn 留了个口子，但是没找到具体实现
     */
    while (hasMoreToolCalls || pendingMessages.length > 0) {
      if (!firstTurn) {
        // 完成一项任务需要持续好几轮 LLM 调用
        await emit({ type: 'turn_start' });
      } else {
        firstTurn = false;
      }
      // 在下一条 Assistent 响应前，插入 steerMessages
      if (pendingMessages.length > 0) {
        for (const message of pendingMessages) {
          await emit({ type: 'message_start', message });
          await emit({ type: 'message_end', message });
          // 赶紧往里塞，干预任务执行
          currentContext.messages.push(message);
          newMessages.push(message);
        }
        pendingMessages = [];
      }
      // 调用 LLM，拿到 Assistent 响应
      const message = await streamAssistantResponse(
        currentContext,
        config,
        signal,
        emit,
        streamFunction
      );
      newMessages.push(message);
      // 如果有停止原因，触发相应钩子
      if (message.stopReason === 'error' || message.stopReason === 'aborted') {
        await emit({ type: 'turn_end', message, toolResults: [] }); // 当前这一轮「LLM + 工具调用」结束
        await emit({ type: 'agent_end', messages: newMessages }); // Agent 完成处理
        return;
      }
      // 获取工具调用消息
      const toolCalls = message.content.filter(c => c.type === 'toolCall');
      const toolResults: ToolResultMessage[] = [];
      hasMoreToolCalls = false;
      if (toolCalls.length > 0) {
        const executedToolBatch =
          message.stopReason === 'length'
            ? /**
               * 失败原因是 length，就表明 token 超限
               * toolCalls 可能被截断导致消息不完整
               * 应该整个废弃掉（添加 isError:true,terminate:false）
               * 没有真正的调用工具，错误信息保留在 messages 中
               * 等待下一轮 LLM 决定是否调用
               */
              await failToolCallsFromTruncatedMessage(toolCalls, emit)
            : /**
               * 执行工具调用，拿到结果
               * 触发 tool_execution_start 钩子函数
               * 通过 executePreparedToolCall 执行工具
               * 调用 finalizeExecutedToolCall 对执行结果进行包装
               * 通过 shouldTerminateToolBatch 判断所有工具是否执行完毕
               * ------------------------------------------------------
               * 工具可以串行和并行执行（只要存在 executionMode 为 sequential 的工具）
               * 无论是 executeToolCallsSequential 还是 executeToolCallsParallel
               * 内部都有一个 immediate 参数：
               * * 这里不是「立即执行」的意思
               * * 而是「立即得到结果，无需真正执行 tool」
               * prepareToolCall 只是做一些参数校验工作
               * executePreparedToolCall 是真正执行工具的地方
               */
              await executeToolCalls(
                currentContext,
                message,
                config,
                signal,
                emit
              );
        toolResults.push(...executedToolBatch.messages);
        hasMoreToolCalls = !executedToolBatch.terminate;
        for (const result of toolResults) {
          currentContext.messages.push(result);
          newMessages.push(result);
        }
      }
      // 当前这一轮「LLM + 工具调用」结束
      await emit({ type: 'turn_end', message, toolResults });
      // 准备下一轮调用的上下文
      const nextTurnContext = {
        message,
        toolResults,
        context: currentContext,
        newMessages
      };
      // 下一轮调用前，允许外部修改下一轮要用的 context、model、thinkingLevel
      const nextTurnSnapshot = await config.prepareNextTurn?.(nextTurnContext);
      if (nextTurnSnapshot) {
        currentContext = nextTurnSnapshot.context ?? currentContext;
        config = {
          ...config,
          model: nextTurnSnapshot.model ?? config.model,
          reasoning:
            nextTurnSnapshot.thinkingLevel === undefined
              ? config.reasoning
              : nextTurnSnapshot.thinkingLevel === 'off'
                ? undefined
                : nextTurnSnapshot.thinkingLevel
        };
      }
      if (
        /**
         * 似乎是为未来留的口子，源码中没找到具体实现
         * true，代表循环结束，触发 agent_end 钩子
         * 不会有下一轮 LLM 调用了
         * 不检查 followUpQueue
         */
        await config.shouldStopAfterTurn?.({
          message,
          toolResults,
          context: currentContext,
          newMessages
        })
      ) {
        // Agent 完成处理
        await emit({ type: 'agent_end', messages: newMessages });
        return;
      }
      // 更新 pendingMessages，直到 steerMessages 消耗干净
      pendingMessages = (await config.getSteeringMessages?.()) || [];
    }
    /**
     * 已经跳出内部的 while 循环，表明工具调用和 steerMessages 已经处理结束
     * 相当于，用户触发的某次任务执行已经结束
     * 查看排队的任务消息列表，存在就继续执行
     */
    const followUpMessages = (await config.getFollowUpMessages?.()) || [];
    if (followUpMessages.length > 0) {
      // 内部 while 继续下一个任务执行
      pendingMessages = followUpMessages;
      continue;
    }
    // 没任何待执行任务，跳出外部 while
    break;
  }
  // Agent 完成处理
  await emit({ type: 'agent_end', messages: newMessages });
}

async function streamAssistantResponse(
  /**
   * 系统提示词
   * 消息历史 + 最新消息
   * 可用工具列表
   */
  context: AgentContext,
  /**
   * 模型配置
   * AgentMessage2Message 转换工具
   * 上下文优化 / 修建工具
   * ...
   */
  config: AgentLoopConfig,
  signal: AbortSignal | undefined,
  emit: AgentEventSink,
  // EventStream 工具函数
  streamFunction: StreamFn
): Promise<AssistantMessage> {
  let messages = context.messages;
  if (config.transformContext) {
    /**
     * 上下文窗口管理：修建旧的消息，例如：
     * * transformContext: async (messages) => {
     * *   if (estimateTokens(messages) > MAX_TOKENS) {
     * *     return pruneOldMessages(messages);
     * *   }
     * *   return messages;
     * * }
     * 将外部资源注入上下文：
     * * 注入系统提示
     * * 添加隐藏消息
     * * 过滤敏感内容
     */
    messages = await config.transformContext(messages, signal);
  }
  /**
   * LLM 只接受 UserMessage、AssistantMessage、ToolResultMessage，例如：
   * convertToLlm: (messages) => messages.flatMap(m => {
   *   if (m.role === "custom") {
   *     return [{ role: "user", content: m.content, timestamp: m.timestamp }];
   *   }
   *   if (m.role === "notification") {
   *     return [];
   *   }
   *   return [m];
   * })
   */
  const llmMessages = await config.convertToLlm(messages);
  // 给 LLM 的上下文
  const llmContext: Context = {
    systemPrompt: context.systemPrompt,
    messages: llmMessages,
    tools: context.tools
  };
  const resolvedApiKey =
    (config.getApiKey
      ? await config.getApiKey(config.model.provider)
      : undefined) || config.apiKey;
  /**
   * packages/ai/src/compat.ts
   * packages/ai/src/providers 身份 / 配置层
   * packages/ai/src/api       协议 / 传输层，具体实现
   * -------------------------------------------------
   * streamFunction 是由各个 ModelProvider 提供的：
   *   Agent.streamFunction → ModelRuntime.streamSimple →
   *   Provider.streamSimple → API.streamSimple → 实际 SDK 调用
   * ----------------------------------------------------------
   * streamSimple 内部都调用了各自的 stream 工具函数
   */
  const response = await streamFunction(config.model, llmContext, {
    ...config,
    apiKey: resolvedApiKey,
    signal
  });
  /**
   * 和常见的前端 SSE 来拼接消息体不同
   * partialMessage / finalMessage 总是「截止到目前」的完整消息快照
   * 当然 LLM 本身返回的还是碎片化的消息，stream 工具函数做了处理：
   * * block.text += event.delta.text;
   * * block.thinking += event.delta.thinking;
   * * block.partialJson += event.delta.partial_json;
   */
  let partialMessage: AssistantMessage | null = null;
  let addedPartial = false;
  /**
   * 返回值 response 是一个 AsyncIterable（AssistantMessageEventStream）
   * Providers 的 streamSimple 会将 AssistantMessage 往 Queue 里不断 push
   * 可以参考 packages/ai/src/api/anthropic-messages.ts 的 stream 工具函数的实现
   */
  for await (const event of response) {
    switch (event.type) {
      case 'start':
        partialMessage = event.partial;
        context.messages.push(partialMessage);
        addedPartial = true;
        await emit({ type: 'message_start', message: { ...partialMessage } });
        break;
      case 'text_start':
      case 'text_delta':
      case 'text_end':
      case 'thinking_start':
      case 'thinking_delta':
      case 'thinking_end':
      case 'toolcall_start':
      case 'toolcall_delta':
      case 'toolcall_end':
        if (partialMessage) {
          partialMessage = event.partial;
          context.messages[context.messages.length - 1] = partialMessage;
          await emit({
            type: 'message_update',
            assistantMessageEvent: event,
            message: { ...partialMessage }
          });
        }
        break;
      case 'done':
      case 'error': {
        const finalMessage = await response.result();
        if (addedPartial) {
          context.messages[context.messages.length - 1] = finalMessage;
        } else {
          context.messages.push(finalMessage);
        }
        if (!addedPartial) {
          await emit({ type: 'message_start', message: { ...finalMessage } });
        }
        await emit({ type: 'message_end', message: finalMessage });
        return finalMessage;
      }
    }
  }
  const finalMessage = await response.result();
  if (addedPartial) {
    context.messages[context.messages.length - 1] = finalMessage;
  } else {
    context.messages.push(finalMessage);
    await emit({ type: 'message_start', message: { ...finalMessage } });
  }
  await emit({ type: 'message_end', message: finalMessage });
  // 最终，完整消息被返回
  return finalMessage;
}
